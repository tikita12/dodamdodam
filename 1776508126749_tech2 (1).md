## 🔒 Firestore Security Rules (⭐ 이 가이드의 핵심)

> **Functions가 없으므로 Rules가 유일한 보안 장치다.**
> AI가 Rules를 잘못 짜면 누구나 모든 데이터를 읽고 쓸 수 있다.

### 필수 패턴: "본인 것만" Rules

**`firestore.rules`**
```
rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // todos 컬렉션 — 본인이 만든 것만 읽기/쓰기
    match /todos/{todoId} {
      // 읽기: 본인 것만
      allow read: if request.auth != null
                  && resource.data.uid == request.auth.uid;

      // 생성: 로그인 + uid 본인 것 + 필수 필드 검증
      allow create: if request.auth != null
                    && request.resource.data.uid == request.auth.uid
                    && request.resource.data.title is string
                    && request.resource.data.title.size() > 0
                    && request.resource.data.title.size() <= 200
                    && request.resource.data.done is bool;

      // 수정: 본인 것만 + uid 변경 금지
      allow update: if request.auth != null
                    && resource.data.uid == request.auth.uid
                    && request.resource.data.uid == resource.data.uid;

      // 삭제: 본인 것만
      allow delete: if request.auth != null
                    && resource.data.uid == request.auth.uid;
    }

    // 그 외 모든 경로 차단 (default deny)
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
```

### Rules 작성 체크리스트 (AI 필독)
- [ ] `request.auth != null` 로그인 확인
- [ ] `resource.data.uid == request.auth.uid` 소유권 확인
- [ ] create 시 필수 필드 타입·길이 검증
- [ ] update 시 uid 변경 금지
- [ ] 마지막에 `match /{document=**} { allow read, write: if false; }` (default deny)
- [ ] **새 컬렉션 추가 시 반드시 Rules도 같이 작성**

### Rules 테스트 (선택)
배포 전 Firebase Emulator로 검증:
```bash
firebase emulators:start --only firestore
```

## 🛠️ Firestore CRUD 헬퍼

### `src/lib/firestore-helpers.ts`
> **이 파일의 함수를 무조건 재사용. AI가 매번 새로 짜면 안 됨.**

```typescript
import {
  doc, getDoc, setDoc, updateDoc, deleteDoc,
  collection, query, where, orderBy, getDocs, onSnapshot,
  serverTimestamp, addDoc,
  type DocumentData, type QueryConstraint,
} from 'firebase/firestore'
import { db } from './firebase'

// ============ READ ============

export async function getDocument<T>(
  path: string
): Promise<(T & { id: string }) | null> {
  const snap = await getDoc(doc(db, path))
  if (!snap.exists()) return null
  return { id: snap.id, ...snap.data() } as T & { id: string }
}

export async function getCollection<T>(
  collectionPath: string,
  ...constraints: QueryConstraint[]
): Promise<(T & { id: string })[]> {
  const q = query(collection(db, collectionPath), ...constraints)
  const snap = await getDocs(q)
  return snap.docs.map(d => ({ id: d.id, ...d.data() })) as (T & { id: string })[]
}

// ============ REALTIME ============

export function subscribeCollection<T>(
  collectionPath: string,
  callback: (items: (T & { id: string })[]) => void,
  ...constraints: QueryConstraint[]
): () => void {
  const q = query(collection(db, collectionPath), ...constraints)
  return onSnapshot(q, (snap) => {
    callback(snap.docs.map(d => ({ id: d.id, ...d.data() })) as (T & { id: string })[])
  })
}

// ============ CREATE ============

export async function addDocument<T extends DocumentData>(
  collectionPath: string,
  data: T
): Promise<string> {
  const ref = await addDoc(collection(db, collectionPath), {
    ...data,
    createdAt: serverTimestamp(),
  })
  return ref.id
}

// ============ UPDATE ============

export async function updateDocument<T extends DocumentData>(
  path: string,
  data: Partial<T>
): Promise<void> {
  await updateDoc(doc(db, path), data)
}

// ============ DELETE ============

export async function deleteDocument(path: string): Promise<void> {
  await deleteDoc(doc(db, path))
}
```

### 사용 예시 (`src/hooks/useTodos.ts`)
```typescript
import { useEffect, useState } from 'react'
import { where, orderBy } from 'firebase/firestore'
import { subscribeCollection, addDocument, updateDocument, deleteDocument } from '@/lib/firestore-helpers'
import { getCollectionPath } from '@/lib/firestore-schema'
import { useAuth } from './useAuth'
import type { TodoWithId, CreateTodoInput, UpdateTodoInput } from '@/types/todo'

export function useTodos() {
  const { user } = useAuth()
  const [todos, setTodos] = useState<TodoWithId[]>([])
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    if (!user) { setTodos([]); setLoading(false); return }

    // 본인 것만 실시간 구독 (Rules와 일치해야 함)
    const unsub = subscribeCollection<TodoWithId>(
      getCollectionPath.todos(),
      (items) => { setTodos(items); setLoading(false) },
      where('uid', '==', user.uid),
      orderBy('createdAt', 'desc'),
    )
    return unsub
  }, [user])

  async function addTodo(input: CreateTodoInput) {
    if (!user) throw new Error('로그인 필요')
    await addDocument(getCollectionPath.todos(), {
      ...input,
      uid: user.uid,        // ⚠️ 반드시 본인 uid (Rules가 검증)
      done: false,
    })
  }

  async function updateTodo(id: string, input: UpdateTodoInput) {
    await updateDocument(getCollectionPath.todo(id), input)
  }

  async function removeTodo(id: string) {
    await deleteDocument(getCollectionPath.todo(id))
  }

  return { todos, loading, addTodo, updateTodo, removeTodo }
}
```

### `src/hooks/useAuth.ts`
```typescript
import { useEffect, useState } from 'react'
import { onAuthStateChanged, type User } from 'firebase/auth'
import { auth } from '@/lib/firebase'

export function useAuth() {
  const [user, setUser] = useState<User | null>(null)
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    const unsub = onAuthStateChanged(auth, (u) => {
      setUser(u)
      setLoading(false)
    })
    return unsub
  }, [])

  return { user, loading }
}
```

## 🛡️ 에러 처리 표준

```typescript
import { FirebaseError } from 'firebase/app'

try {
  setLoading(true)
  await addTodo({ title })
} catch (error) {
  if (error instanceof FirebaseError) {
    if (error.code === 'permission-denied') {
      setError('권한이 없습니다 (Rules 확인)')
    } else {
      setError('Firebase 오류: ' + error.code)
    }
  } else if (error instanceof Error) {
    setError(error.message)
  } else {
    setError('알 수 없는 오류')
  }
} finally {
  setLoading(false)
}
```

> **자주 만나는 에러**:
> - `permission-denied` → Firestore Rules가 동작 차단 → Rules 검토
> - `unauthenticated` → 로그인 안 된 상태에서 요청 → 보호 라우트 누락
